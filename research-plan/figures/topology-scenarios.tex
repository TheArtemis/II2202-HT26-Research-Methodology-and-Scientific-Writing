\newcommand{\pentagoncoords}[1]{%
  \coordinate (#1A) at (-0.42,0.06);
  \coordinate (#1B) at (0,0.40);
  \coordinate (#1C) at (0.42,0.06);
  \coordinate (#1D) at (-0.26,-0.34);
  \coordinate (#1E) at (0.26,-0.34);
}

\newcommand{\alllinks}[1]{%
  \foreach \u/\v in {A/B,A/C,A/D,A/E,B/C,B/D,B/E,C/D,C/E,D/E}
    {\draw[live] (#1\u)--(#1\v);}%
}

\newcommand{\fivenodes}[2]{%
  \node[replica] at (#1A) {A};
  \node[replica] at (#1B) {B};
  \node[#2]      at (#1C) {C};
  \node[replica] at (#1D) {D};
  \node[replica] at (#1E) {E};
}

\newcommand{\routedvia}[4]{%
  \draw[detour]
    (#1#2) to[bend right=16] (#1#3)
           to[bend right=16] (#1#4);
}

\newcommand{\miniarrow}{%
  \draw[-{Stealth[length=1.2mm]},draw=panelgray,line width=0.7pt]
    (-0.12,0)--(0.12,0);
}

% Title above cell; graphs only inside; notes well below cell.
% #1 origin  #2 title  #3 tag  #4 left note  #5 right note  #6 left  #7 right
\newcommand{\topocell}[7]{%
  \begin{scope}[shift={#1}]
    \node[title] at (-2.70,1.85) {#2};
    \node[tag]   at ( 2.70,1.85) {#3};
    % Tall cell: empty padding under graphs so captions never collide.
    \draw[cell] (-2.70,-1.05) rectangle (2.70,1.58);
    \node[colhead] at (-1.25,1.40) {direct};
    \node[colhead] at ( 1.25,1.40) {forwarding};
    \begin{scope}[shift={(-1.25,0.35)}]#6\end{scope}%
    \begin{scope}[shift={(0,0.35)}]\miniarrow\end{scope}%
    \begin{scope}[shift={(1.25,0.35)}]#7\end{scope}%
    \node[note] at (-1.25,-1.32) {#4};
    \node[note] at ( 1.25,-1.32) {#5};
  \end{scope}
}
